Example shape (do not use this verbatim — adapt to match the existing file's patterns):
```typescript
export async function adminGetAllOrders(): Promise<Order[]> {
  const token = getAuthToken();
  const apiBase = import.meta.env.VITE_API_URL || '/api';
  const res = await fetch(`${apiBase}/orders/owned`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  if (!res.ok) throw new Error(`adminGetAllOrders: ${res.status}`);
  const json = await res.json();
  return json.data ?? json;
}
```
